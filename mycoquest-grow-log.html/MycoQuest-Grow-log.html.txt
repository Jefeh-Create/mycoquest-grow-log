<script>
const KEY = "mycoquest-grow-log-v2";
const BACKUP_KEY = "mycoquest-grow-log-v2-invalid-backup";
const CAP = 50;
const DEMO = { growStage: 100, growPhoto: 250, firstGrow: 250, firstPost: 100 };
const MULT = { supporting: 1, "co-star": 1.5, star: 2 };
const STAGES = ["Planning", "Inoculation", "Colonization", "Fruiting", "Harvest", "Complete"];
const STATUSES = ["pending", "approved", "rejected"];

/* ---------- Level curve ---------- */
// Cost to go from level L to L+1.
// Levels 1–45: blocks of 5 level-ups; block N costs 1,000 × N, so each level-up costs 200 × N.
// Levels 46–49 (block 10, only 4 level-ups): 2,500 each, so block 10 totals 10,000.
function levelUpCost(L) {
  if (L >= CAP) return 0;
  if (L >= 46) return 2500;
  return 200 * Math.ceil(L / 5);
}
// Total XP needed to reach level L.
function xpToReachLevel(L) {
  let t = 0;
  for (let l = 1; l < L; l++) t += levelUpCost(l);
  return t;
}
function levelFromXp(xp) {
  let L = 1;
  while (L < CAP && xp >= xpToReachLevel(L + 1)) L++;
  return L;
}

/* ---------- State ---------- */
let state = load();
let editingId = null;
let previewUrl = "";

function emptyState() { return { version: 2, entries: [], recipes: [] }; }

function isValidState(s) {
  if (!s || typeof s !== "object" || s.version !== 2 || !Array.isArray(s.entries) || !Array.isArray(s.recipes)) return false;
  const str = (x) => typeof x === "string";
  const num = (x) => Number.isFinite(x) && x >= 0;
  const ids = [...s.entries, ...s.recipes].map((x) => x && x.id);
  if (new Set(ids).size !== ids.length) return false;
  const entriesOk = s.entries.every((e) => e && str(e.id) && str(e.growName) && str(e.species) &&
    STAGES.includes(e.stage) && str(e.date) && str(e.observation) && str(e.createdAt) &&
    STATUSES.includes(e.status) && num(e.xpAwarded ?? 0) && num(e.proposedXp ?? 0));
  const recipesOk = s.recipes.every((r) => r && str(r.id) && str(r.recipeName) && str(r.createdAt) &&
    STATUSES.includes(r.status) && Number.isFinite(r.complexity) && r.complexity >= 1 && r.complexity <= 5 &&
    MULT[r.role] !== undefined && num(r.xpAwarded ?? 0) && num(r.proposedXp ?? 0));
  return entriesOk && recipesOk;
}

function normalizeState(s) {
  return {
    version: 2,
    entries: s.entries.map((e) => ({
      rejectionReason: "", photoName: "", proposedXp: 0, xpAwarded: 0, ...e,
      stageCredited: e.stageCredited === true, photoCredited: e.photoCredited === true,
      firstGrowCredited: e.firstGrowCredited === true, firstPostCredited: e.firstPostCredited === true
    })),
    recipes: s.recipes.map((r) => ({
      rejectionReason: "", proposedXp: 0, xpAwarded: 0, hasVideo: false, ...r,
      hasVideo: r.hasVideo === true, firstPostCredited: r.firstPostCredited === true
    }))
  };
}

function load() {
  let raw = null;
  try {
    raw = localStorage.getItem(KEY);
    if (!raw) return emptyState();
    const parsed = JSON.parse(raw);
    if (isValidState(parsed)) return normalizeState(parsed);
    localStorage.setItem(BACKUP_KEY, raw);
    flash("#backupMsg", "Saved data was not in the expected format. A copy was kept under a backup key; use Import to restore a file.");
  } catch (err) {
    flash("#backupMsg", "Could not read saved data in this browser. Use Import to restore a backup.");
  }
  return emptyState();
}

function save() {
  try {
    localStorage.setItem(KEY, JSON.stringify(state));
  } catch (err) {
    flash("#backupMsg", "Browser storage is unavailable. Export your data now to keep it.");
  }
}

/* ---------- Helpers ---------- */
const $ = (s) => document.querySelector(s);
const norm = (s) => String(s).toLocaleLowerCase().replace(/\s+/g, " ").trim();
const fmt = (n) => Number(n).toLocaleString("en-US", { maximumFractionDigits: 2 });
const round2 = (n) => Math.round(n * 100) / 100;
const uid = (p) => `${p}-${Date.now().toString(36)}-${Math.random().toString(36).slice(2, 7)}`;
const nowIso = () => new Date().toISOString();
function todayKey() {
  const d = new Date();
  return `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, "0")}-${String(d.getDate()).padStart(2, "0")}`;
}
function flash(sel, text) { const n = $(sel); if (n) n.textContent = text; }
function el(tag, props = {}, ...kids) {
  const n = document.createElement(tag);
  Object.assign(n, props);
  kids.forEach((k) => n.append(k));
  return n;
}
function btn(text, onclick, cls = "") {
  return el("button", { type: "button", textContent: text, className: cls, onclick });
}

function totalXp(s = state) {
  return s.entries.reduce((a, e) => a + e.xpAwarded, 0) + s.recipes.reduce((a, r) => a + r.xpAwarded, 0);
}

/* ---------- Duplicate and milestone checks (evaluated at the moment of approval) ---------- */
function stageKey(e) { return `${norm(e.growName)}::${norm(e.stage)}`; }

function credited() {
  const approved = state.entries.filter((e) => e.status === "approved");
  return {
    stages: new Set(approved.filter((e) => e.stageCredited).map(stageKey)),
    photos: new Set(approved.filter((e) => e.photoCredited).map(stageKey)),
    firstGrow: approved.some((e) => e.firstGrowCredited),
    firstPost: approved.some((e) => e.firstPostCredited) ||
      state.recipes.some((r) => r.status === "approved" && r.firstPostCredited)
  };
}

function growEstimate(e) {
  const c = credited();
  const k = stageKey(e);
  const stage = c.stages.has(k) ? 0 : DEMO.growStage;
  const photo = e.photoName && !c.photos.has(k) ? DEMO.growPhoto : 0;
  const firstGrow = c.firstGrow ? 0 : DEMO.firstGrow;
  const firstPost = c.firstPost ? 0 : DEMO.firstPost;
  return { stage, photo, firstGrow, firstPost, total: stage + photo + firstGrow + firstPost, duplicate: stage === 0 };
}

function recipeBase(r) {
  return round2(r.complexity * 55 * MULT[r.role] + (r.hasVideo ? 100 : 0));
}
function recipeEstimateNow(r) {
  return round2(recipeBase(r) + (credited().firstPost ? 0 : DEMO.firstPost));
}

function approveGrow(e) {
  const x = growEstimate(e);
  Object.assign(e, {
    status: "approved", xpAwarded: x.total,
    stageCredited: x.stage > 0, photoCredited: x.photo > 0,
    firstGrowCredited: x.firstGrow > 0, firstPostCredited: x.firstPost > 0,
    decidedAt: nowIso()
  });
  return x;
}

function approveRecipe(r) {
  const bonus = credited().firstPost ? 0 : DEMO.firstPost;
  Object.assign(r, {
    status: "approved", proposedXp: recipeBase(r),
    xpAwarded: round2(recipeBase(r) + bonus), firstPostCredited: bonus > 0, decidedAt: nowIso()
  });
  return bonus;
}

function decide(kind, id, action, reason = "") {
  const list = kind === "grow" ? state.entries : state.recipes;
  const item = list.find((x) => x.id === id);
  if (!item || item.status !== "pending") return;
  let text;
  if (action === "reject") {
    Object.assign(item, {
      status: "rejected", rejectionReason: reason, xpAwarded: 0,
      stageCredited: false, photoCredited: false, firstGrowCredited: false, firstPostCredited: false,
      decidedAt: nowIso()
    });
    text = "Rejected. 0 XP awarded (demo only).";
  } else if (kind === "grow") {
    const x = approveGrow(item);
    text = `Approved (demo only). +${fmt(x.total)} XP.` +
      (x.duplicate ? " This stage was already credited, so no stage XP was added." : "");
  } else {
    const bonus = approveRecipe(item);
    text = `Approved (demo only). +${fmt(item.xpAwarded)} XP.` +
      (bonus ? "" : " First-post bonus was already used.");
  }
  save();
  renderAll();
  flash(kind === "grow" ? "#growMsg" : "#recipeMsg", text);
}

function removePending(kind, id) {
  const list = kind === "grow" ? state.entries : state.recipes;
  const item = list.find((x) => x.id === id);
  if (!item || item.status !== "pending") return;
  if (!confirm("Delete this pending post? It has not awarded any XP.")) return;
  if (kind === "grow") state.entries = state.entries.filter((x) => x.id !== id);
  else state.recipes = state.recipes.filter((x) => x.id !== id);
  if (editingId === id) exitEditMode();
  save();
  renderAll();
  flash(kind === "grow" ? "#growMsg" : "#recipeMsg", "Pending post deleted.");
}

/* ---------- Rendering ---------- */
function renderLevel() {
  const total = totalXp();
  const L = levelFromXp(total);
  const start = xpToReachLevel(L);
  $("#level").textContent = `Level ${L}`;
  if (L >= CAP) {
    $("#levelCaption").textContent = "Level 50 cap reached";
    $("#levelBar").style.width = "100%";
    $("#xpLine").textContent = `${fmt(total)} total demo XP`;
  } else {
    const next = xpToReachLevel(L + 1);
    const into = total - start;
    const need = next - start;
    $("#levelBar").style.width = `${Math.min(100, (into / need) * 100)}%`;
    $("#levelCaption").textContent = `${fmt(need - into)} XP to Level ${L + 1}`;
    $("#xpLine").textContent = `${fmt(into)} / ${fmt(need)} XP into this level · ${fmt(total)} total`;
  }
  const all = [...state.entries, ...state.recipes];
  const count = (s) => all.filter((x) => x.status === s).length;
  $("#counts").textContent = `${count("pending")} pending · ${count("approved")} approved · ${count("rejected")} rejected`;
}

function pendingControls(item, kind, msgSel) {
  const box = el("div", { className: "actions" });
  if (kind === "grow") box.append(btn("Edit", () => startEdit(item.id), "ghost"));
  box.append(btn("Delete", () => removePending(kind, item.id), "ghost danger"));
  box.append(btn("Approve · demo only", () => decide(kind, item.id, "approve")));
  const reason = el("input", { placeholder: "Rejection reason", maxLength: 200 });
  box.append(reason, btn("Reject · demo only", () => {
    const r = reason.value.trim();
    if (!r) { flash(msgSel, "Enter a rejection reason first."); return; }
    decide(kind, item.id, "reject", r);
  }, "ghost"));
  return box;
}

function growCard(e) {
  const card = el("div", { className: "entry" });
  card.append(
    el("div", { textContent: `${e.date} · ${e.stage} · ${e.species}` }),
    el("div", { className: "muted", textContent: e.observation })
  );
  if (e.photoName) card.append(el("div", { className: "muted", textContent: `Photo filename: ${e.photoName} (keep the original file; photos are not exported)` }));
  if (e.status === "pending") {
    const est = growEstimate(e);
    card.append(el("div", {
      className: "status pending",
      textContent: `Pending review · 0 XP awarded · estimated ${fmt(est.total)} demo XP if approved` +
        (est.duplicate ? " (stage already credited, so 0 stage XP)" : "")
    }));
    card.append(pendingControls(e, "grow", "#growMsg"));
  } else if (e.status === "approved") {
    card.append(el("div", { className: "status approved", textContent: `Approved · ${fmt(e.xpAwarded)} demo XP awarded` }));
  } else {
    card.append(el("div", { className: "status rejected", textContent: "Rejected · 0 XP" }));
    card.append(el("div", { className: "muted", textContent: `Reason: ${e.rejectionReason}` }));
  }
  return card;
}

function renderGrow() {
  const box = $("#growList");
  box.replaceChildren();
  if (!state.entries.length) { box.append(el("p", { className: "muted", textContent: "No grow posts yet." })); return; }
  const groups = new Map();
  [...state.entries].sort((a, b) => a.createdAt.localeCompare(b.createdAt)).forEach((e) => {
    const k = norm(e.growName);
    if (!groups.has(k)) groups.set(k, []);
    groups.get(k).push(e);
  });
  for (const list of groups.values()) {
    const g = el("div", { className: "group" });
    g.append(el("h3", { textContent: `${list[0].growName} · ${list[0].species}` }));
    list.forEach((e) => g.append(growCard(e)));
    box.append(g);
  }
}

function recipeCard(r) {
  const card = el("div", { className: "entry" });
  card.append(el("div", { textContent: `${r.recipeName} · complexity ${fmt(r.complexity)} · ${r.role}${r.hasVideo ? " · video" : ""}` }));
  if (r.status === "pending") {
    card.append(el("div", { className: "status pending", textContent: `Pending review · 0 XP awarded · estimated ${fmt(recipeEstimateNow(r))} demo XP if approved` }));
    card.append(pendingControls(r, "recipe", "#recipeMsg"));
  } else if (r.status === "approved") {
    card.append(el("div", { className: "status approved", textContent: `Approved · ${fmt(r.xpAwarded)} demo XP awarded` }));
  } else {
    card.append(el("div", { className: "status rejected", textContent: "Rejected · 0 XP" }));
    card.append(el("div", { className: "muted", textContent: `Reason: ${r.rejectionReason}` }));
  }
  return card;
}

function renderRecipes() {
  const box = $("#recipeList");
  box.replaceChildren();
  if (!state.recipes.length) { box.append(el("p", { className: "muted", textContent: "No recipe posts yet." })); return; }
  state.recipes.forEach((r) => box.append(recipeCard(r)));
}

function renderAll() {
  renderLevel();
  renderGrow();
  renderRecipes();
}

/* ---------- Grow form ---------- */
const growForm = $("#growForm");
const photoInput = $("#photo");
const preview = $("#photoPreview");

function clearPhotoPreview() {
  if (previewUrl) URL.revokeObjectURL(previewUrl);
  previewUrl = "";
  preview.removeAttribute("src");
  preview.hidden = true;
}

photoInput.addEventListener("change", () => {
  clearPhotoPreview();
  const f = photoInput.files[0];
  if (!f) return;
  if (!f.type.startsWith("image/") || f.size > 25 * 1024 * 1024) {
    flash("#growMsg", "Choose an image up to 25 MB.");
    photoInput.value = "";
    return;
  }
  previewUrl = URL.createObjectURL(f);
  preview.src = previewUrl;
  preview.hidden = false;
});

function startEdit(id) {
  const item = state.entries.find((x) => x.id === id);
  if (!item || item.status !== "pending") return;
  editingId = id;
  ["growName", "species", "stage", "date", "observation"].forEach((k) => { growForm.elements[k].value = item[k]; });
  $("#formTitle").textContent = "Edit pending grow post";
  $("#growSubmit").textContent = "Save changes · stays pending";
  $("#growCancel").hidden = false;
  growForm.scrollIntoView({ behavior: "smooth" });
}

function exitEditMode() {
  editingId = null;
  $("#formTitle").textContent = "Log grow activity";
  $("#growSubmit").textContent = "Submit grow post · pending review";
  $("#growCancel").hidden = true;
}

$("#growCancel").addEventListener("click", () => {
  exitEditMode();
  growForm.reset();
  growForm.elements.date.value = todayKey();
  clearPhotoPreview();
  flash("#growMsg", "Edit canceled.");
});

growForm.addEventListener("submit", (ev) => {
  ev.preventDefault();
  const f = new FormData(growForm);
  const data = {
    growName: String(f.get("growName")).trim(),
    species: String(f.get("species")).trim(),
    stage: String(f.get("stage")),
    date: String(f.get("date")),
    observation: String(f.get("observation")).trim()
  };
  if (!data.growName || !data.species || !data.observation || !STAGES.includes(data.stage) ||
      !/^\d{4}-\d{2}-\d{2}$/.test(data.date)) {
    flash("#growMsg", "Fill in grow name, species, stage, date, and observation.");
    return;
  }
  const file = photoInput.files[0];
  if (file && (!file.type.startsWith("image/") || file.size > 25 * 1024 * 1024)) {
    flash("#growMsg", "Choose an image up to 25 MB.");
    return;
  }
  let msg;
  if (editingId) {
    const item = state.entries.find((x) => x.id === editingId);
    if (item && item.status === "pending") {
      Object.assign(item, data, { updatedAt: nowIso() });
      if (file) item.photoName = file.name;
      msg = "Changes saved. Still pending review; 0 XP.";
    }
    exitEditMode();
  } else {
    state.entries.push({
      id: uid("grow"), ...data, photoName: file ? file.name : "",
      status: "pending", rejectionReason: "", xpAwarded: 0, proposedXp: 0,
      createdAt: nowIso(),
      stageCredited: false, photoCredited: false, firstGrowCredited: false, firstPostCredited: false
    });
    msg = "Submitted. Pending review; 0 XP until approved.";
  }
  save();
  growForm.reset();
  growForm.elements.date.value = todayKey();
  clearPhotoPreview();
  renderAll();
  flash("#growMsg", msg);
});

/* ---------- Recipe form ---------- */
$("#recipeForm").addEventListener("submit", (ev) => {
  ev.preventDefault();
  const f = new FormData(ev.currentTarget);
  const complexity = Number(f.get("complexity"));
  const role = String(f.get("role"));
  const recipeName = String(f.get("recipeName")).trim();
  if (!recipeName || !Number.isFinite(complexity) || complexity < 1 || complexity > 5 || !MULT[role]) {
    flash("#recipeMsg", "Enter a recipe name, a complexity from 1.0 to 5.0, and a mushroom role.");
    return;
  }
  const r = {
    id: uid("recipe"), recipeName, complexity, role, hasVideo: f.get("video") === "yes",
    status: "pending", rejectionReason: "", xpAwarded: 0, proposedXp: 0,
    createdAt: nowIso(), firstPostCredited: false
  };
  state.recipes.push(r);
  save();
  ev.currentTarget.reset();
  renderAll();
  flash("#recipeMsg", `Submitted. Estimated ${fmt(recipeEstimateNow(r))} demo XP if approved; 0 XP now.`);
});

/* ---------- Backup: export, import, reset ---------- */
$("#exportBtn").addEventListener("click", () => {
  const payload = {
    app: "MycoQuest grow log", version: 2, demoOnly: true, exportedAt: nowIso(),
    totalDemoXp: totalXp(), level: levelFromXp(totalXp()),
    entries: state.entries, recipes: state.recipes
  };
  const url = URL.createObjectURL(new Blob([JSON.stringify(payload, null, 2)], { type: "application/json" }));
  const a = el("a", { href: url, download: `mycoquest-grow-log-${todayKey()}.json` });
  document.body.append(a);
  a.click();
  a.remove();
  setTimeout(() => URL.revokeObjectURL(url), 1000);
  flash("#backupMsg", `Exported ${state.entries.length} grow posts and ${state.recipes.length} recipe posts.`);
});

$("#importFile").addEventListener("change", async (ev) => {
  const file = ev.target.files[0];
  ev.target.value = "";
  if (!file) return;
  try {
    const data = JSON.parse(await file.text());
    if (data.app !== "MycoQuest grow log" || data.version !== 2) {
      throw new Error("this is not a MycoQuest grow log export (version 2).");
    }
    const candidate = { version: 2, entries: data.entries, recipes: data.recipes };
    if (!isValidState(candidate)) throw new Error("the file is missing required fields or has invalid values.");
    const next = normalizeState(candidate);
    const ok = confirm(
      `Replace your current data (${state.entries.length} grow, ${state.recipes.length} recipe posts, ` +
      `${fmt(totalXp())} XP) with the file (${next.entries.length} grow, ${next.recipes.length} recipe posts, ` +
      `${fmt(totalXp(next))} XP)? Export first if you are not sure.`
    );
    if (!ok) { flash("#backupMsg", "Import canceled. Nothing changed."); return; }
    state = next;
    save();
    renderAll();
    flash("#backupMsg", `Imported. Totals recalculated from entries: Level ${levelFromXp(totalXp())}, ${fmt(totalXp())} demo XP.`);
  } catch (err) {
    flash("#backupMsg", `Import failed: ${err.message} Nothing changed.`);
  }
});

$("#resetBtn").addEventListener("click", () => {
  if (!confirm("Delete all grow log and recipe data saved in this browser? Export first if you want a backup.")) return;
  state = emptyState();
  exitEditMode();
  save();
  renderAll();
  flash("#backupMsg", "Demo data cleared on this device.");
});
